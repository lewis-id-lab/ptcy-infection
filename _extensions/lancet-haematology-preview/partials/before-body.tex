$-- Lancet-style opening block: masthead, full-width title, authors, summary.
$-- In twocolumn mode \twocolumn[...] is what spans both columns, so everything
$-- that must run full width has to sit inside these brackets.
\twocolumn[%
  \begin{@twocolumnfalse}
  \thispagestyle{lancetfirst}
  \vspace*{-4mm}
  {\sffamily\bfseries\Large\lancetmasthead{Articles}\par}
  \vspace{1.5mm}
  {\color{lancetred}\rule{\textwidth}{0.7pt}}
  \vspace{7mm}

  \lancettitle{$title$}
$if(by-author)$
  \lancetauthors{$for(by-author)$$it.name.literal$$sep$, $endfor$}
$endif$

$if(abstract)$
  \lancetsummaryhead
  {\small $abstract$}
$endif$
  \vspace{5mm}
  \end{@twocolumnfalse}
]
